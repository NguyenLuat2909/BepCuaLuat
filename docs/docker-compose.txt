version: "3.9"

services:
  app:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: bep-cua-luat
    restart: unless-stopped
    ports:
      - "3000:3000"
    environment:
      NODE_ENV: production
      DATABASE_URL: ${DATABASE_URL}
      AUTH_SECRET: ${AUTH_SECRET}
      AUTH_URL: ${AUTH_URL}
      NEXT_PUBLIC_CREATE_APP_URL: ${NEXT_PUBLIC_CREATE_APP_URL}
      GEMINI_API_KEY: ${GEMINI_API_KEY}
      NEXT_PUBLIC_UPLOADCARE_PUBLIC_KEY: ${NEXT_PUBLIC_UPLOADCARE_PUBLIC_KEY}
    depends_on:
      - postgres
    networks:
      - bep-network

  postgres:
    image: postgres:17-alpine
    container_name: bep-postgres
    restart: unless-stopped
    environment:
      POSTGRES_USER: bepuser
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-changeme}
      POSTGRES_DB: bepcualuat
    volumes:
      - bep-pgdata:/var/lib/postgresql/data
      - ./database/schema.sql:/docker-entrypoint-initdb.d/01-schema.sql:ro
    ports:
      - "5432:5432"
    networks:
      - bep-network

volumes:
  bep-pgdata:

networks:
  bep-network:
    driver: bridge
